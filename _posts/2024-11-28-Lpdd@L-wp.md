```python
import socket
import re
import time

# 设置服务器信息
HOST = 'challenge.hubuctf.cn'
# 服务器地址
PORT = 31382

# 端口号
def solve_math_question(question):
# 提取数学表达式中的两个数字
match = re.search(r"(\d+)\s*([+\-*/])\s*(\d+)", question)
if match:
    num1 = int(match.group(1))
    operator = match.group(2)
    num2 = int(match.group(3))
    if operator == '+':
        return num1 + num2
    elif operator == '-':
        return num1 - num2
    elif operator == '*':
        return num1 * num2
    elif operator == '/':
        return num1 // num2 # 整数除法
return None

def main():
# 创建与服务器的连接
      with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
             s.connect((HOST, PORT))
            while True:
# 从服务器读取问题
                data = s.recv(1024).decode()
                print(f"Received from server: {data}")

# 打印出从服务器接收到的内容
# 检查是否包含 flag flag_match = re.search(r'flag\{.*\}',data)

                if flag_match:
                    print(f"Flag found: {flag_match.group()}")
                    break

# 如果问题没有“Question”就停止
                if "Question" not in data:
                    print("No more questions.")
                    break

# 提取问题并计算答案
question = re.search(r'Question \d+: (.+?) =', data)
if question:
     question_text = question.group(1)
# 获取数学题文本
    answer = solve_math_question(question_text)
    if answer is not None:
        print(f"Answering: {answer}")
# 发送答案
        s.sendall(f"{answer}\n".encode())

# 增加一个非常小的等待时间来避免过多请求导致超时
time.sleep(0.1)# 只需要非常短的时间间隔

if __name__ == '__main__':
    main()

```

<mark>***一些base的联想和范围点：***</mark>

- 只有字母大小写和数字，联想到 base62
- 字母大小写和 2~7，联想到 base58
- 大写字母，数字，少数特殊字符，联想到 base45
- 大写字母，4 个=，联想到 base32

（<mark>注意一下，base全家桶解密赛博厨子会更加全面一点，随波逐流只能起到辅助作用，</mark>
<mark>因为随波逐流没有base45，它和一般网站一样缺少一些不太常见的base，所以要注意</mark>）

-  Base16使用16个ASCII可打印字符（数字0-9和字母A-F）
-  Base32编码是使用32个可打印字符（字母A-Z和数字2-7
- 相比Base64，Base58不使用数字"0"，字母大写"O"，字母大写"I"，和字母小写"l"，以及"+"和"/"符号。
-  Base64中的可打印字符包括字母A-Z、a-z、数字0-9，这样共有62个字符
- 从94个可打印ASCII字符（0x21-0x7E）中，以下三个字符被省略以构建basE91字母：
　　　　- -（破折号，0x2D），　\（反斜杠，0x5C），　'（撇号，0x27）
