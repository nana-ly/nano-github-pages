三、**实验总结**

在本次实验的初期实现过程中，我遇到了传统逆序数算法无法正确处理相同元素的问题。原版的归并排序逆序计数算法在使用严格小于关系进行比较时，会忽略相同元素之间的逆序关系，导致计数结果不准确。通过深入分析算法原理，我发现问题的关键在于比较条件的选择，将原来的严格小于关系改为小于等于关系，这样既保证了排序的稳定性，又确保了相同元素不会被错误地计入逆序数中。

在程序调试过程中，我遇到了几个关键问题。最初测试包含重复元素的序列时，发现逆序数统计结果与预期存在偏差。经过仔细分析，发现问题出现在mergeAndCount方法的比较逻辑上。当left[i]与right[j]相等时，应该优先移动左指针而不进行逆序数统计，这样才能正确反映重复元素间的顺序关系。通过单步调试和对比测试用例，验证了修改后的算法能够准确处理各种包含重复元素的序列情况。

通过本次实验，我深刻认识到算法设计与实现的细节重要性。一个看似微小的比较条件修改，实际上关系到整个算法的正确性和适用性。同时，我也体会到分治思想在解决复杂问题时的强大威力，归并排序不仅能够高效地进行排序，还能在排序过程中统计逆序数，这种算法设计思路值得在未来的编程实践中借鉴和应用。

这次实践不仅加深了我对逆序数计算原理的理解及分治策略的应用，更培养了我对算法细节的把握能力和调试技巧，为今后解决更复杂的算法问题积累了经验。

import java.util.Scanner;

public class BMC {

    /**

     * 分治法实现二进制数乘法

     * @param x 第一个二进制数（字符串形式）

     * @param y 第二个二进制数（字符串形式）

     * @return 乘积的二进制字符串

     */

    public static String multiplyBinary(String x, String y) {

        // 确保两个字符串长度相同，便于处理

        int n = Math.max(x.length(), y.length());

        x = padZeros(x, n);

        y = padZeros(y, n);

        // 基础情况

        if (n == 1) {

            return (x.charAt(0) <mark> '1' && y.charAt(0) </mark> '1') ? "1" : "0";

        }

        int mid = n / 2;

        // 分割字符串

        String a = x.substring(0, mid);    // 高半部分

        String b = x.substring(mid);       // 低半部分

        String c = y.substring(0, mid);    // 高半部分

        String d = y.substring(mid);       // 低半部分

        // 递归计算

        String ac = multiplyBinary(a, c);      // a × c

        String bd = multiplyBinary(b, d);      // b × d

        String abcd = multiplyBinary(addBinary(a, b), addBinary(c, d)); // (a+b) × (c+d)

        // 计算 ad + bc = (a+b)(c+d) - ac - bd

        String adPlusBc = subtractBinary(subtractBinary(abcd, ac), bd);

        // 结果 = ac × 2^n + adPlusBc × 2^(n/2) + bd

        String term1 = shiftLeft(ac, 2 * (n - mid));  // ac × 2^n

        String term2 = shiftLeft(adPlusBc, n - mid);  // (ad+bc) × 2^(n/2)

        return addBinary(addBinary(term1, term2), bd);

    }

    /**

     * 二进制加法

     */

    private static String addBinary(String a, String b) {

        StringBuilder result = new StringBuilder();

        int i = a.length() - 1, j = b.length() - 1;

        int carry = 0;

        while (i >= 0 || j >= 0 || carry > 0) {

            int sum = carry;

            if (i >= 0) sum += a.charAt(i--) - '0';

            if (j >= 0) sum += b.charAt(j--) - '0';

            result.append(sum % 2);

            carry = sum / 2;

        }

        return result.reverse().toString();

    }

    /**

     * 二进制减法（假设a >= b）

     */

    private static String subtractBinary(String a, String b) {

        // 简化实现，假设a >= b

        a = removeLeadingZeros(a);

        b = removeLeadingZeros(b);

        if (b.equals("0")) return a;

        if (a.equals(b)) return "0";

        // 补齐长度

        int maxLen = Math.max(a.length(), b.length());

        a = padZeros(a, maxLen);

        b = padZeros(b, maxLen);

        StringBuilder result = new StringBuilder();

        int borrow = 0;

        for (int i = maxLen - 1; i >= 0; i--) {

            int digitA = a.charAt(i) - '0' - borrow;

            int digitB = b.charAt(i) - '0';

            if (digitA < digitB) {

                digitA += 2;

                borrow = 1;

            } else {

                borrow = 0;

            }

            result.append(digitA - digitB);

        }

        return removeLeadingZeros(result.reverse().toString());

    }

    /**

     * 左移（相当于乘以2的n次方）

     */

    private static String shiftLeft(String binary, int n) {

        return binary + "0".repeat(n);

    }

    /**

     * 补零到指定长度

     */

    private static String padZeros(String binary, int length) {

        if (binary.length() >= length) return binary;

        return "0".repeat(length - binary.length()) + binary;

    }

    /**

     * 去除前导零

     */

    private static String removeLeadingZeros(String binary) {

        int i = 0;

        while (i < binary.length() - 1 && binary.charAt(i) == '0') {

            i++;

        }

        return binary.substring(i);

    }

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        System.out.print("请输入第一个二进制数: ");

        String binary1 = scanner.nextLine();

        System.out.print("请输入第二个二进制数: ");

        String binary2 = scanner.nextLine();

        // 验证输入是否为合法二进制数

        if (!binary1.matches("[01]+") || !binary2.matches("[01]+")) {

            System.out.println("错误：请输入有效的二进制数（只包含0和1）");

            return;

        }

        String result = multiplyBinary(binary1, binary2);

        System.out.println("乘法结果: " + result);

        scanner.close();

    }

}
