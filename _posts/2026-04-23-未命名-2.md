数组倒置啊，每行每列相加减去交叉点，求最大值
class Solution:
    def maxSumAfterReverse(self, grid):
        n = len(grid)
        m = len(grid[0])
        
        def rev(x):
            return int(str(x)[::-1])           数组倒置可以先定这个倒置函数
        
        max_val = 0
        
        for i in range(n):           # 倒置第 i 行
            for k in range(m):       # 选第 k 列
                total = 0
                
                for r in range(n):   # 遍历每一行
                    for c in range(m):   # 把这行的所有数加上
                        if r == i:
                            total += rev(grid[r][c])
                        else:
                            total += grid[r][c]
                    
                    # 把第 k 列的数再加一次
                    if r == i:
                        total += rev(grid[r][k])
                    else:
                        total += grid[r][k]
                    
                    # 减掉第 k 列（因为被加了两次）
                    if r == i:
                        total -= rev(grid[r][k])
                    else:
                        total -= grid[r][k]
                
                if total > max_val:
                    max_val = total
        
        return max_val

**当你需要保留原来的头节点时**，才会额外定义 `cur`。

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

class Solution:
    def processList(self, head: ListNode) -> ListNode:
        # 1. 空链表判断
        if not head:
            return None
        
        # 2. 计算链表长度，同时找到第二个节点
        length = 0
        second = None
        cur = head
        while cur:
            length += 1
            if length == 2:
                second = cur       # 记住第二个节点
            cur = cur.next
        
        # 3. 长度不足 3，直接返回原链表
        if length < 3:
            return head
        
        # 4. 把第二个节点的值复制一份，追加到链表尾部
        cur = head
        while cur.next:
            cur = cur.next
        cur.next = ListNode(second.val)
        
        # 5. 删除前三个元素（新头是第 4 个节点）
        new_head = head
        for i in range(3):          # 用 i 代替 _
            new_head = new_head.next
        
        return new_head

将两个升序链表合并为一个新的 **升序** 链表并返回。新链表是通过拼接给定的两个链表的所有节点组成的。
class ListNode:

    def __init__(self, val=0, next=None):

        self.val = val

        self.next = next

  

class Solution:

    def mergeTwoLists(self, list1: Optional[ListNode], list2: Optional[ListNode]) -> Optional[ListNode]:

        #先判断空，有一个链表为空直接新链表就是另一个链表

        if list1==None:

            return list2

        if list2==None:

            return list1

        #更小的值来放在头部，小于等于都可以

        if list1.val<=list2.val:

            head=list1

            list1=list1.next  #节点往后移动

        else:

            head=list2

            list2=list2.next

        #新链表的节点

        cur=head

  

        #继续遍历两链表后面的，小的值接在后面

        while list1 and list2:

            if list1.val<=list2.val:

                cur.next=list1 #新链表接小值，是节点接上这个

                list1=list1.next  #节点往后移动

            else:

                cur.next=list2

                list2=list2.next

            cur=cur.next

  

        #遍历完所有较小的值还会有剩余的没接上去，剩余的直接加在链表后面

        if list1!=None:

            cur.next=list1

        if list2!=None:

            cur.next=list2

        return head
